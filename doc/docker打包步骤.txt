# 构建代码镜像
docker compose -f docker-compose.my.yml build

## 镜像导出
docker save -o dist/ruoyi-all.tar ruoyi-backend-my:latest ruoyi-frontend:latest mysql:8.0 redis:latest

## 镜像导入
docker load -i ruoyi-all.tar



# 部署运行

## 打开终端
在 E:\plat\PayloadGroundTest 下运行终端

## 启动服务
docker compose -f docker-compose.my.yml up -d

## 停止服务
docker compose -f docker-compose.my.yml down
